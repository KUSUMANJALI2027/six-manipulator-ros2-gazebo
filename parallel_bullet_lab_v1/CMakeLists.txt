cmake_minimum_required(VERSION 3.16)
project(ParallelBulletLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Bullet REQUIRED)
add_executable(parallel_core src/main.cc)
target_include_directories(parallel_core PRIVATE ${BULLET_INCLUDE_DIRS})
target_link_libraries(parallel_core PRIVATE ${BULLET_LIBRARIES})
option(BUILD_GAZEBO_PLUGIN "Build the Gazebo Harmonic system plugin" ON)
if(BUILD_GAZEBO_PLUGIN)
 find_package(gz-sim8 REQUIRED)
 find_package(gz-plugin2 REQUIRED COMPONENTS register)
 add_library(ParallelBulletSystem SHARED src/ParallelSystem.cc)
 target_include_directories(ParallelBulletSystem PRIVATE ${BULLET_INCLUDE_DIRS})
 target_link_libraries(ParallelBulletSystem PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 ${BULLET_LIBRARIES})
endif()
